cmake_minimum_required(VERSION 3.20)
project(YudreamUiNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_THIRDPARTY_CONTAINERS OFF CACHE BOOL "" FORCE)
set(RMLUI_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
find_package(JNI REQUIRED)
add_subdirectory("${RMLUI_SOURCE}" rmlui EXCLUDE_FROM_ALL)
add_library(yudreamui SHARED bridge.cpp)
target_include_directories(yudreamui PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(yudreamui PRIVATE RmlUi::Core)
if(NOT MSVC)
    target_compile_options(yudreamui PRIVATE -ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)
endif()
